### 2.1 CMake 配置模块

#### 2.1.1 新建 cmake/optional/CangjieMimalloc.cmake

```cmake
# Copyright (c) Huawei Technologies Co., Ltd. 2025. All rights reserved.
# This source file is part of the Cangjie project, licensed under Apache-2.0
# with Runtime Library Exception.
#
# See https://cangjie-lang.cn/pages/LICENSE for license information.

# CangjieMimalloc.cmake
# Configures mimalloc as the default memory allocator for the Cangjie compiler.
#
# Options:
#   CANGJIE_USE_MIMALLOC - Enable mimalloc (default: OFF)
#
# Exported variables (when enabled):
#   CANGJIE_MIMALLOC_STATIC_LIB  - Path to mimalloc static library
#   CANGJIE_MIMALLOC_INCLUDE_DIR - Path to mimalloc headers

if(NOT CANGJIE_USE_MIMALLOC)
    return()
endif()

# ============================================================
# Conflict detection
# ============================================================

set(MIMALLOC_CONFLICT_REASON "")

if(CANGJIE_SANITIZER_SUPPORT_ENABLED)
    set(MIMALLOC_CONFLICT_REASON "sanitizer support (${CANGJIE_SANITIZER_SUPPORT})")
elseif(CANGJIE_ENABLE_ASAN_COV)
    set(MIMALLOC_CONFLICT_REASON "ASAN coverage (CANGJIE_ENABLE_ASAN_COV)")
elseif(CANGJIE_ENABLE_COMPILER_TSAN)
    set(MIMALLOC_CONFLICT_REASON "compiler TSAN (CANGJIE_ENABLE_COMPILER_TSAN)")
endif()

if(MIMALLOC_CONFLICT_REASON)
    message(WARNING
        "CANGJIE_USE_MIMALLOC is automatically disabled because it is "
        "incompatible with ${MIMALLOC_CONFLICT_REASON}.\n"
        "mimalloc and memory sanitizers cannot be used simultaneously.")
    set(CANGJIE_USE_MIMALLOC OFF CACHE BOOL
        "Use mimalloc as the default memory allocator" FORCE)
    return()
endif()

# ============================================================
# Platform-specific strategy selection
# ============================================================

# On Windows, use new/delete override strategy to avoid DLL boundary issues.
# On Linux/macOS, use global malloc/free override for maximum coverage.
if(WIN32)
    set(CANGJIE_MIMALLOC_STRATEGY "new-delete"
        CACHE STRING "mimalloc integration strategy: global|new-delete")
    message(STATUS "mimalloc strategy: new-delete (Windows DLL boundary safety)")
else()
    set(CANGJIE_MIMALLOC_STRATEGY "global"
        CACHE STRING "mimalloc integration strategy: global|new-delete")
    message(STATUS "mimalloc strategy: global (Linux/macOS full replacement)")
endif()

# ============================================================
# Find or build mimalloc
# ============================================================

set(MIMALLOC_SOURCE_DIR ${CMAKE_SOURCE_DIR}/third_party/mimalloc)
set(MIMALLOC_BUILD_DIR ${CMAKE_BINARY_DIR}/third_party/mimalloc)

if(NOT EXISTS ${MIMALLOC_SOURCE_DIR}/CMakeLists.txt)
    message(FATAL_ERROR
        "mimalloc source not found at ${MIMALLOC_SOURCE_DIR}.\n"
        "Please run: git submodule update --init third_party/mimalloc")
endif()

# Configure mimalloc build options
set(MI_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(MI_BUILD_STATIC ON CACHE BOOL "" FORCE)
set(MI_BUILD_OBJECT ON CACHE BOOL "" FORCE)
set(MI_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(MI_INSTALL_TOPLEVEL OFF CACHE BOOL "" FORCE)

# Enable override based on strategy
if(CANGJIE_MIMALLOC_STRATEGY STREQUAL "global")
    set(MI_OVERRIDE ON CACHE BOOL "" FORCE)
else()
    set(MI_OVERRIDE OFF CACHE BOOL "" FORCE)
endif()

# Security settings
if(CMAKE_BUILD_TYPE MATCHES "^(Debug|RelWithDebInfo)$")
    set(MI_SECURE ON CACHE BOOL "" FORCE)
else()
    set(MI_SECURE OFF CACHE BOOL "" FORCE)
endif()

add_subdirectory(${MIMALLOC_SOURCE_DIR} ${MIMALLOC_BUILD_DIR} EXCLUDE_FROM_ALL)

# Export variables for use in src/CMakeLists.txt
set(CANGJIE_MIMALLOC_STATIC_LIB mimalloc-static CACHE INTERNAL "")
set(CANGJIE_MIMALLOC_INCLUDE_DIR ${MIMALLOC_SOURCE_DIR}/include CACHE INTERNAL "")

message(STATUS "mimalloc: ENABLED (version ${MI_VERSION}, strategy: ${CANGJIE_MIMALLOC_STRATEGY})")
```

/**
 * @file MimallocNewDelete.h
 *
 * Enables mimalloc as the C++ operator new/delete implementation.
 * Include this file ONCE in each executable's main translation unit.

#endif

### 3.3 性能基准测试脚本

```bash
#!/bin/bash
# scripts/benchmark_mimalloc.sh

set -e

PROJECTS=("small_project" "medium_project" "large_project")
ITERATIONS=5

echo "date,project,allocator,time_s,memory_mb" > benchmark_results.csv

for project in "${PROJECTS[@]}"; do
    for allocator in "system" "mimalloc"; do
        # 构建对应版本
        if [ "$allocator" = "mimalloc" ]; then
            cmake -DCANGJIE_USE_MIMALLOC=ON .. > /dev/null 2>&1
        else
            cmake -DCANGJIE_USE_MIMALLOC=OFF .. > /dev/null 2>&1
        fi
        cmake --build . -j$(nproc) > /dev/null 2>&1

        # 运行多次取平均
        total_time=0
        total_memory=0
        for i in $(seq 1 $ITERATIONS); do
            result=$(./cjc --profile-compile-time --profile-compile-memory \
                          test_projects/$project/ 2>&1)
            time=$(echo "$result" | grep "Total:" | awk '{print $2}' | tr -d 's')
            memory=$(echo "$result" | grep "Peak Memory:" | awk '{print $3}' | tr -d 'MB')
            total_time=$(echo "$total_time + $time" | bc)
            total_memory=$(echo "$total_memory + $memory" | bc)
        done

        avg_time=$(echo "scale=3; $total_time / $ITERATIONS" | bc)
        avg_memory=$(echo "scale=1; $total_memory / $ITERATIONS" | bc)

        echo "$(date +%Y-%m-%d),$project,$allocator,$avg_time,$avg_memory" \
            >> benchmark_results.csv
    done
done

# 生成对比报告
python3 scripts/generate_perf_report.py benchmark_results.csv
```

## 6. 后续优化方向

### 6.1 短期（1-3个月）

**方案 C 补充**：针对热点数据结构使用 `mi_heap_t` 隔离堆
```cpp
// 编译单元级别的隔离堆
class CompilationUnit {
    mi_heap_t* astHeap = mi_heap_new();
    mi_heap_t* chirHeap = mi_heap_new();

    ~CompilationUnit() {
        mi_heap_delete(astHeap);   // 批量释放所有 AST 内存
        mi_heap_delete(chirHeap);  // 批量释放所有 CHIR 内存
    }
};
```

### 6.2 中期（3-6个月）

**增量编译优化**：利用 mimalloc 的内存快照功能
```cpp
// 保存编译状态快照
mi_heap_t* snapshot = mi_heap_new();
// ... 编译 ...
// 如果增量编译失败，恢复快照
mi_heap_delete(snapshot);
```

### 6.3 长期（6-12个月）

**自定义 Arena 整合**：将现有的 `InterpreterArena` 迁移到 mimalloc
```cpp
// 现有 Arena 实现（InterpreterArena.h）
// 可以替换为 mimalloc 的 mi_heap_t，获得更好的性能
class Arena {
    mi_heap_t* heap = mi_heap_new();

    IVal* Allocate(IVal&& value) {
        void* mem = mi_heap_malloc(heap, sizeof(IVal));
        return new(mem) IVal(std::move(value));
    }

    ~Arena() {
        mi_heap_delete(heap);  // 一次性释放所有内存
    }
};
```
