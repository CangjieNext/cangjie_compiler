### 5.1 P3 级优化实施（Week 1）

**BUG-21：抽取共享头文件**

```cpp
// src/cangjie/Utils/MimallocInit.h
#pragma once

#ifdef CANGJIE_USE_MIMALLOC
#include <mimalloc.h>

#if defined(_WIN32)
#include <mimalloc-new-delete.h>

// 使用编译器属性确保初始化顺序优先级
#  if defined(__GNUC__) || defined(__clang__)
struct __attribute__((init_priority(101))) MimallocInit {
    MimallocInit() { mi_process_init(); }
};
#  else
struct MimallocInit {
    MimallocInit() { mi_process_init(); }
};
#  endif
static MimallocInit g_mimallocInit;
#endif

static inline void InitMimallocOptions()
{
    mi_option_set(mi_option_reset_delay, 0);
    mi_option_set(mi_option_eager_commit_delay, 0);
}

static inline void CheckMimallocInit()
{
    void* test = mi_malloc(16);
    if (!test) {
        fprintf(stderr, "Warning: mimalloc initialization failed.\n");
    } else {
        mi_free(test);
    }
}

#endif // CANGJIE_USE_MIMALLOC
```

**入口点简化**：

```cpp
// src/main.cpp 及其他入口点
#include "cangjie/Utils/MimallocInit.h"

int main(int argc, const char** argv, const char** envp)
{
#ifdef CANGJIE_USE_MIMALLOC
    InitMimallocOptions();
    CheckMimallocInit();
#endif
    // ...
}
```
